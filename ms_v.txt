#Volumes

Command	Description
docker volume ls	Lists all Docker volumes.
docker volume create myvol	Creates a volume named myvol.
docker volume inspect myvol	Displays details about the volume.
docker volume rm myvol	Removes a volume if it is not in use.
docker volume prune	Removes unused anonymous and named volumes.
docker volume prune -a	Removes all unused volumes, including named volumes.
docker run -d --name c1 -v myvol:/data nginx	Runs Nginx with myvol mounted at /data.
docker run -d --name c1 --mount source=myvol,target=/data nginx	Mounts myvol at /data using explicit mount syntax.
docker inspect c1	Displays container configuration, including volume mounts.
docker inspect -f "{{json .Mounts}}" c1	Displays the container's mount details.
docker run --rm -v myvol:/data alpine ls /data	Lists files in the volume from a temporary Alpine container.
docker run --rm -v myvol:/data -v "$PWD":/backup alpine cp -r /data/. /backup/	Copies volume contents to a host directory on compatible shells.
docker stop c1	Stops the container using the volume.
docker rm c1	Removes the container but preserves a separately created named volume.
docker run --rm -v myvol:/data alpine sh -c "echo Hello > /data/file.txt"	Writes a file into the volume.
docker run --rm -v myvol:/data alpine cat /data/file.txt	Reads the file stored in the volume.

#network

Command	One-line description
docker network ls	Lists all Docker networks.
docker network create mynet	Creates a user-defined bridge network.
docker network inspect mynet	Displays network details and connected containers.
docker network rm mynet	Removes a network when no containers are connected.
docker network connect mynet c1	Connects an existing container to a network.
docker network disconnect mynet c1	Disconnects a container from a network.
docker network prune	Removes unused networks.
docker inspect c1	Displays the container's configuration, including networking.
docker port c1	Displays published host-to-container port mappings.
docker exec c1 hostname -i	Displays the container's IP address or addresse


Step 1: Create a network
docker network create mynet


Step 2: Start the web server
docker run -d --name c1 --network mynet nginx:alpine


Step 3: Start the client
docker run -dit --name c2 --network mynet alpine:latest sh


Step 4: Send an HTTP request from c2 to c1
docker exec c2 wget -qO- http://c1


Displays the HTML response returned by Nginx.
Step 5: Check the network
docker network inspect mynet


Shows both containers connected to mynet.
Step 6: Test interactively
docker exec -it c2 sh


Inside the client container, execute:
wget -qO- http://c1


Type exit to leave the shell.

Command	Description
docker exec client ping server	Pings the server container from the client.
docker exec -it client ping -c 4 server	Sends 4 ping packets to the server.
docker exec -it client ping 8.8.8.8	Tests connectivity to a public IP address.
docker exec -it client ping google.com	Tests external connectivity and DNS resolution.
docker exec -it server ping client	Pings the client from the server container.
docker exec -it client sh	Opens a shell inside the client container.
docker network inspect my-network	Checks the containers connected to the network.




Command	Description
docker network create my-network	Creates a custom bridge network named my-network.
docker run -d --name server --network my-network nginx:alpine	Starts an NGINX server container in the network.
docker run -dit --name client --network my-network busybox:1.36	Starts a BusyBox client container in interactive background mode.
docker ps	Lists running containers.
docker exec client wget -qO- http://server	Sends an HTTP request from the client to the server using its container name.
docker logs server	Displays the NGINX server logs.
docker rm -f client server	Forcefully stops and removes both containers.
docker network rm my-network	Removes the custom network after its containers are removed.
