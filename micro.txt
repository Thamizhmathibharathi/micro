#node.js
//
// Filename: server.js
// Simple Node.js HTTP Server
//

const http = require('http');

// Port number
const PORT = process.env.PORT || 8888;

// Create HTTP server
http.createServer((request, response) => {
    response.end('Server is running.!! You asked for: ' + request.url);
}).listen(PORT, () => {
    console.log("Server listening on: http://localhost:%s", PORT);
});


# Use the official Node.js image as the base image
FROM node

# Set the working directory
WORKDIR /usr/src

# Copy the server code into the image
COPY server.js .

# Document the port used by the server
EXPOSE 8888

# Start the server when the container runs
CMD ["node", "./server.js"]\

web-stack.yaml

apiVersion: apps/v1
kind: Deployment
metadata:
  name: frontend-deployment
spec:
  replicas: 2
  selector:
    matchLabels:
      tier: frontend
  template:
    metadata:
      labels:
        tier: frontend
    spec:
      containers:
        - name: nginx
          image: nginx:1.27-alpine
          ports:
            - name: http
              containerPort: 80
              protocol: TCP
---
apiVersion: v1
kind: Service
metadata:
  name: frontend-service
spec:
  type: ClusterIP
  selector:
    tier: frontend
  ports:
    - name: http
      port: 80
      targetPort: http
      protocol: TCP
---
apiVersion: apps/v1
kind: Deployment

  name: backend-deployment
